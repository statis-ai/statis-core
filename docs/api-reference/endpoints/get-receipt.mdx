---
title: "Get Receipt"
api: "GET /receipts/{action_id}"
description: "Fetch the tamper-evident SHA-256 receipt for a completed or denied action."
---

## Path parameters

<ParamField path="action_id" type="string" required>
  The action whose receipt to retrieve.
</ParamField>

## Response

<ResponseField name="receipt_id" type="string" />
<ResponseField name="action_id" type="string" />
<ResponseField name="decision" type="string">`APPROVED` or `DENIED`</ResponseField>
<ResponseField name="rule_id" type="string | null">The policy rule that fired.</ResponseField>
<ResponseField name="rule_version" type="string | null" />
<ResponseField name="approved_by" type="string">`policy_engine` or `human:{reviewer_id}`</ResponseField>
<ResponseField name="conditions_evaluated" type="object | null">Per-condition pass/fail detail.</ResponseField>
<ResponseField name="entity_state_snapshot" type="object | null">Full entity state at evaluation time.</ResponseField>
<ResponseField name="execution_result" type="object | null">Adapter output. Includes reviewer metadata if escalated.</ResponseField>
<ResponseField name="executed_at" type="string | null" />
<ResponseField name="hash" type="string">SHA-256 of canonical receipt fields.</ResponseField>
<ResponseField name="created_at" type="string" />
