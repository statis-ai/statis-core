---
title: "Get Action"
api: "GET /actions/{action_id}"
description: "Get the current status and details of an action contract."
---

## Path parameters

<ParamField path="action_id" type="string" required>
  The action identifier.
</ParamField>

## Response

<ResponseField name="action_id" type="string" />
<ResponseField name="status" type="string">
  One of: `PROPOSED`, `APPROVED`, `DENIED`, `ESCALATED`, `EXECUTING`, `COMPLETED`, `FAILED`
</ResponseField>
<ResponseField name="action_type" type="string" />
<ResponseField name="target_entity" type="object" />
<ResponseField name="parameters" type="object" />
<ResponseField name="proposed_by" type="string" />
<ResponseField name="target_system" type="string" />
<ResponseField name="context" type="object" />
<ResponseField name="created_at" type="string" />
<ResponseField name="updated_at" type="string" />
